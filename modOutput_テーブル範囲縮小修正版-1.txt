Private Sub SetCardTitle( _
    ByVal ws As Worksheet, _
    ByVal cardTopLeft As String, _
    ByVal titleText As String)

    Dim currentValue As String
    Dim p As Long

    currentValue = CStr(ws.Range(cardTopLeft).value)

    p = InStr(currentValue, vbLf)

    If p > 0 Then
        currentValue = Mid$(currentValue, p + 1)
    Else
        currentValue = "-"
    End If

    ws.Range(cardTopLeft).value = _
        titleText & vbLf & currentValue

End Sub


'==========================================================
' ボタン作成
'==========================================================
Private Sub CreateButtonAt( _
    ByVal ws As Worksheet, _
    ByVal leftPos As Double, _
    ByVal topPos As Double, _
    ByVal buttonWidth As Double, _
    ByVal buttonHeight As Double, _
    ByVal caption As String, _
    ByVal macroName As String, _
    ByVal fillColor As Long)

    Dim shp As shape

    Set shp = _
        ws.Shapes.AddShape( _
            msoShapeRoundedRectangle, _
            leftPos, _
            topPos, _
            buttonWidth, _
            buttonHeight)

    With shp
        .Fill.ForeColor.RGB = fillColor
        .line.Visible = msoFalse

        .TextFrame2.TextRange.Text = caption
        .TextFrame2.TextRange.Font.Size = 10
        .TextFrame2.TextRange.Font.Bold = msoTrue
        .TextFrame2.TextRange.Font.Fill.ForeColor.RGB = vbWhite

        .TextFrame2.VerticalAnchor = msoAnchorMiddle
        .TextFrame2.TextRange.ParagraphFormat.Alignment = _
            msoAlignCenter

        .OnAction = macroName

        .Placement = xlFreeFloating
    End With

End Sub


'==========================================================
' Shape削除
'==========================================================
Private Sub DeleteToolShapes( _
    ByVal ws As Worksheet)

    Dim i As Long

    For i = ws.Shapes.Count To 1 Step -1
        ws.Shapes(i).Delete
    Next i

End Sub

'==========================================================
' ウィンドウ枠固定
'
' freezeRow = 16 の場合
'
'   1～16行目 ：固定
'   17行目以降：スクロール
'
' 既存のFreezePanes・Split・スクロール位置を
' 完全に解除してから設定し直す。
'==========================================================
Private Sub ResetSheetWindow( _
    ByVal ws As Worksheet)

    On Error GoTo ExitHandler

    ws.Activate

    With ActiveWindow
        .FreezePanes = False
        .SplitRow = 0
        .SplitColumn = 0
        .ScrollRow = 1
        .ScrollColumn = 1
    End With

    ws.Range("A1").Select

ExitHandler:

End Sub


Private Sub FreezeSheetAt( _
    ByVal ws As Worksheet, _
    ByVal freezeRow As Long)

    Dim targetCell As Range

    On Error GoTo ExitHandler

    ws.Activate

    '------------------------------------------------------
    ' 既存状態を完全リセット
    '------------------------------------------------------
    With ActiveWindow

        .FreezePanes = False

        .SplitRow = 0
        .SplitColumn = 0

        .ScrollRow = 1
        .ScrollColumn = 1

    End With

    '------------------------------------------------------
    ' 17行目を選択
    '
    ' A列を選ぶことで列方向は固定しない。
    ' freezeRow = 16ならA17。
    '------------------------------------------------------
    Set targetCell = _
        ws.Cells(freezeRow + 1, 1)

    targetCell.Select

    '------------------------------------------------------
    ' 選択セルより上を固定
    '------------------------------------------------------
    ActiveWindow.FreezePanes = True

    '------------------------------------------------------
    ' 表示位置を先頭に戻す
    '------------------------------------------------------
    With ActiveWindow
        .ScrollRow = 1
        .ScrollColumn = 1
    End With

ExitHandler:

End Sub


'==========================================================
' 経過月数
'==========================================================
Private Function GetElapsedMonths( _
    ByVal startDate As Date, _
    ByVal endDate As Date) As Long

    Dim months As Long

    If endDate <= startDate Then
        GetElapsedMonths = 0
        Exit Function
    End If

    months = _
        DateDiff( _
            "m", _
            startDate, _
            endDate)

    If Day(endDate) < Day(startDate) Then
        months = months - 1
    End If

    If months < 0 Then
        months = 0
    End If

    GetElapsedMonths = months

End Function

'==========================================================
' FileList 年数フィルターボタン表示更新
'==========================================================
Private Sub RefreshFileListFilterButtonCaptions()

    Dim ws As Worksheet
    Dim shp As Shape
    Dim actionName As String

    Set ws = ThisWorkbook.Worksheets(SHEET_FILELIST)

    For Each shp In ws.Shapes

        actionName = GetShapeMacroName(shp)

        Select Case actionName

            Case "ShowOldFiles1"
                SetShapeCaption _
                    shp, _
                    CStr(gOldYear1) & "年以上"

            Case "ShowOldFiles3"
                SetShapeCaption _
                    shp, _
                    CStr(gOldYear2) & "年以上"

            Case "ShowOldFiles5"
                SetShapeCaption _
                    shp, _
                    CStr(gOldYear3) & "年以上"

        End Select

    Next shp

End Sub


Private Function GetShapeMacroName( _
    ByVal shp As Shape) As String

    Dim actionName As String
    Dim p As Long

    On Error Resume Next
    actionName = CStr(shp.OnAction)
    On Error GoTo 0

    If Len(actionName) = 0 Then Exit Function

    'ブック名付きで返る場合にも対応
    '例: '棚卸し.xlsm'!ShowOldFiles1
    p = InStrRev(actionName, "!")

    If p > 0 Then
        actionName = Mid$(actionName, p + 1)
    End If

    actionName = Replace$(actionName, "'", "")
    actionName = Trim$(actionName)

    GetShapeMacroName = actionName

End Function


Private Sub SetShapeCaption( _
    ByVal shp As Shape, _
    ByVal caption As String)

    On Error Resume Next

    shp.TextFrame2.TextRange.Text = caption

    If Err.Number <> 0 Then
        Err.Clear
        shp.TextFrame.Characters.Text = caption
    End If

    On Error GoTo 0

End Sub


'==========================================================
' FileList出力
'==========================================================
Private Sub OutputFileList()

    Dim ws As Worksheet
    Dim data() As Variant

    Dim fileCount As Long
    Dim r As Long

    Dim sizeBytes As Double
    Dim modifiedDate As Date
    Dim elapsedMonths As Long

    Set ws = ThisWorkbook.Worksheets(SHEET_FILELIST)

    fileCount = GetFileCount()

    DeleteExistingTable ws, TABLE_FILE

    ws.Range( _
        "B17:U" & _
        ws.Rows.Count).ClearContents

    If fileCount = 0 Then
        ws.Range("C13").value = "すべて"
        ws.Range("C14").value = "0 / 0 件"
        Exit Sub
    End If

    If fileCount > ws.Rows.Count - 16 Then
        Err.Raise _
            vbObjectError + 300, _
            , _
            "FileListがExcelの最大行数を超えます。"
    End If

    ReDim data( _
        1 To fileCount, _
        1 To 20)

    For r = 1 To fileCount

        sizeBytes = _
            CDbl(GetFileValue(FC_SIZE, r))

        modifiedDate = _
            CDate(GetFileValue(FC_MODIFIED, r))

        elapsedMonths = _
            GetElapsedMonths( _
                modifiedDate, _
                Now)

        data(r, 1) = _
            GetFileValue(FC_NAME, r)

        data(r, 2) = _
            GetFileValue(FC_CATEGORY, r)

        data(r, 3) = _
            sizeBytes / BYTES_PER_MB

        data(r, 4) = _
            modifiedDate

        data(r, 5) = _
            elapsedMonths

        data(r, 6) = _
            GetFileValue(FC_CLEANUP, r)

        data(r, 7) = _
            GetFileValue(FC_REASON, r)

        data(r, 8) = _
            GetFileValue(FC_DUPLICATE, r)

        data(r, 9) = _
            GetFileValue(FC_LARGE, r)

        data(r, 10) = _
            GetFileValue(FC_PARENT, r)

        data(r, 11) = _
            GetFileValue(FC_LEVEL, r)

        data(r, 12) = _
            GetFileValue(FC_RELATIVE, r)

        data(r, 13) = _
            GetFileValue(FC_FULLPATH, r)

        data(r, 14) = _
            GetFileValue(FC_EXT, r)

        data(r, 15) = _
            sizeBytes

        data(r, 16) = _
            GetFileValue(FC_CREATED, r)

        data(r, 17) = _
            GetFileValue(FC_DAYS, r)

        data(r, 18) = _
            GetFileValue(FC_OLD, r)

        data(r, 19) = _
            GetFileValue(FC_NAMEWARN, r)

        data(r, 20) = _
            GetFileValue(FC_PATHWARN, r)

    Next r

    ws.Range("B17").Resize( _
        fileCount, _
        20).value = data

    'サイズ(MB)
    ws.Columns("D").NumberFormat = _
        "#,##0.00 ""MB"""

    '最終更新日時
    ws.Columns("E").NumberFormat = _
        "yyyy/mm/dd hh:mm"

    '経過月数
    ws.Columns("F").NumberFormat = _
        "0 ""か月"""

    'サイズ(Byte)
    ws.Columns("P").NumberFormat = _
        "#,##0"

    '作成日時
    ws.Columns("Q").NumberFormat = _
        "yyyy/mm/dd hh:mm"

    '経過日数
    ws.Columns("R").NumberFormat = _
        "#,##0 ""日"""

    CreateTable _
        ws, _
        ws.Range("B16").Resize( _
            fileCount + 1, _
            20), _
        TABLE_FILE

    ws.Range("C13").value = "すべて"

    ws.Range("C14").value = _
        Format$(fileCount, "#,##0") & _
        " / " & _
        Format$(fileCount, "#,##0") & _
        " 件"

End Sub


'==========================================================
' FolderSummary出力
'==========================================================
Private Sub OutputFolderSummary()

    Dim ws As Worksheet
    Dim folders As Object
    Dim health As Object

    Dim keys As Variant
    Dim data() As Variant
    Dim folderData As Variant
    Dim healthData As Variant

    Dim i As Long
    Dim rowIndex As Long
    Dim folderCount As Long

    Set ws = ThisWorkbook.Worksheets(SHEET_FOLDER)
    Set folders = GetFolderDictionary()

    DeleteExistingTable ws, TABLE_FOLDER

    ws.Range( _
        "B17:Q" & _
        ws.Rows.Count).ClearContents

    If folders Is Nothing Then Exit Sub
    If folders.Count = 0 Then Exit Sub

    folderCount = folders.Count

    If folderCount > ws.Rows.Count - 15 Then
        Err.Raise _
            vbObjectError + 301, _
            , _
            "FolderSummaryがExcelの最大行数を超えます。"
    End If

    Set health = BuildFolderHealthData(folders)

    ReDim data( _
        1 To folderCount, _
        1 To 16)

    keys = folders.keys

    For i = LBound(keys) To UBound(keys)

        rowIndex = rowIndex + 1
        folderData = folders(keys(i))

        data(rowIndex, 1) = folderData(1)
        data(rowIndex, 2) = folderData(4)
        data(rowIndex, 3) = folderData(5)

        data(rowIndex, 4) = _
            CDbl(folderData(6)) / BYTES_PER_MB

        data(rowIndex, 5) = folderData(7)

        data(rowIndex, 6) = _
            CDbl(folderData(8)) / BYTES_PER_MB

        data(rowIndex, 7) = folderData(15)

        data(rowIndex, 8) = _
            CDbl(folderData(16)) / BYTES_PER_MB

        data(rowIndex, 9) = folderData(9)

        If CBool(folderData(10)) Then
            data(rowIndex, 10) = "○"
        Else
            data(rowIndex, 10) = ""
        End If

        If Not IsEmpty(folderData(11)) Then
            data(rowIndex, 11) = folderData(11)
        End If

        data(rowIndex, 12) = folderData(3)
        data(rowIndex, 13) = folderData(2)

        If health.Exists(CStr(keys(i))) Then
            healthData = health(CStr(keys(i)))

            data(rowIndex, 14) = healthData(1)
            data(rowIndex, 15) = healthData(2)
            data(rowIndex, 16) = healthData(3)
        End If

    Next i

    ws.Range("B16").Resize( _
        folderCount, _
        16).value = data

    ws.Columns("E").NumberFormat = _
        "#,##0.00 ""MB"""

    ws.Columns("G").NumberFormat = _
        "#,##0.00 ""MB"""

    ws.Columns("I").NumberFormat = _
        "#,##0.00 ""MB"""

    ws.Columns("L").NumberFormat = _
        "yyyy/mm/dd hh:mm"

    ws.Columns("O").NumberFormat = _
        "0.0 ""点"""

    CreateTable _
        ws, _
        ws.Range("B15").Resize( _
            folderCount + 1, _
            16), _
        TABLE_FOLDER

    SortFoldersNormal

    ws.Range("C12").value = "すべて"

    ws.Range("C13").value = _
        Format$(folderCount, "#,##0") & _
        " / " & _
        Format$(folderCount, "#,##0") & _
        " フォルダ"

End Sub


'==========================================================
' フォルダ健全度
'
' 対象フォルダ直下だけではなく、配下の全ファイルと
' 全サブフォルダを含めて評価する。
'
' 最大減点（重み）はSettingsで変更可能。
' 初期値：30 / 25 / 15 / 15 / 10 / 5
' 合計100になるよう設定する。

' 各悪化度 = Min(実測率 / 最大悪化基準率, 1)
' 件数率と容量率を使う項目は、項目ごとの比率で合成する。
'==========================================================
Private Function BuildFolderHealthData( _
    ByVal folders As Object) As Object

    Const IDX_FILE_COUNT As Long = 1
    Const IDX_TOTAL_BYTES As Long = 2
    Const IDX_OLD_COUNT As Long = 3
    Const IDX_OLD_BYTES As Long = 4
    Const IDX_CLEANUP_COUNT As Long = 5
    Const IDX_CLEANUP_BYTES As Long = 6
    Const IDX_DUP_COUNT As Long = 7
    Const IDX_DUP_BYTES As Long = 8
    Const IDX_LARGE_OLD_COUNT As Long = 9
    Const IDX_LARGE_OLD_BYTES As Long = 10
    Const IDX_WARN_COUNT As Long = 11
    Const IDX_FOLDER_COUNT As Long = 12
    Const IDX_EMPTY_FOLDER_COUNT As Long = 13

    Const STAT_COUNT As Long = 13

    Const LIMIT_OLD As Double = 0.4
    Const LIMIT_CLEANUP As Double = 0.3
    Const LIMIT_DUP As Double = 0.1
    Const LIMIT_LARGE_OLD As Double = 0.05
    Const LIMIT_WARN As Double = 0.1
    Const LIMIT_EMPTY As Double = 0.3

    Dim stats As Object
    Dim result As Object

    Dim keys As Variant
    Dim key As Variant
    Dim folderData As Variant
    Dim statData As Variant
    Dim parentData As Variant

    Dim i As Long
    Dim level As Long
    Dim maxLevel As Long
    Dim fileCount As Long

    Dim parentPath As String
    Dim fileSize As Double
    Dim yearsOld As Double

    Dim countRatio As Double
    Dim byteRatio As Double
    Dim severityOld As Double
    Dim severityCleanup As Double
    Dim severityDup As Double
    Dim severityLargeOld As Double
    Dim severityWarn As Double
    Dim severityEmpty As Double

    Dim penaltyOld As Double
    Dim penaltyCleanup As Double
    Dim penaltyDup As Double
    Dim penaltyLargeOld As Double
    Dim penaltyWarn As Double
    Dim penaltyEmpty As Double

    Dim score As Double
    Dim rank As String
    Dim reason As String
    Dim healthData(1 To 3) As Variant

    Set stats = CreateObject("Scripting.Dictionary")
    stats.CompareMode = vbTextCompare

    Set result = CreateObject("Scripting.Dictionary")
    result.CompareMode = vbTextCompare

    keys = folders.keys

    '------------------------------------------------------
    ' 全フォルダの初期値
    '------------------------------------------------------
    For Each key In keys

        folderData = folders(key)

        ReDim statData(1 To STAT_COUNT)

        statData(IDX_FILE_COUNT) = 0&
        statData(IDX_TOTAL_BYTES) = 0#
        statData(IDX_OLD_COUNT) = 0&
        statData(IDX_OLD_BYTES) = 0#
        statData(IDX_CLEANUP_COUNT) = 0&
        statData(IDX_CLEANUP_BYTES) = 0#
        statData(IDX_DUP_COUNT) = 0&
        statData(IDX_DUP_BYTES) = 0#
        statData(IDX_LARGE_OLD_COUNT) = 0&
        statData(IDX_LARGE_OLD_BYTES) = 0#
        statData(IDX_WARN_COUNT) = 0&

        '自分自身を1フォルダとして数える。
        statData(IDX_FOLDER_COUNT) = 1&

        If CBool(folderData(10)) Then
            statData(IDX_EMPTY_FOLDER_COUNT) = 1&
        Else
            statData(IDX_EMPTY_FOLDER_COUNT) = 0&
        End If

        stats.Add CStr(key), statData

        If CLng(folderData(4)) > maxLevel Then
            maxLevel = CLng(folderData(4))
        End If

    Next key

    '------------------------------------------------------
    ' ファイルを直下フォルダへ集計
    '------------------------------------------------------
    fileCount = GetFileCount()

    For i = 1 To fileCount

        parentPath = _
            CStr(GetFileValue(FC_PARENT, i))

        If stats.Exists(parentPath) Then

            statData = stats(parentPath)

            fileSize = _
                CDbl(GetFileValue(FC_SIZE, i))

            yearsOld = _
                CDbl(GetFileValue(FC_YEARS, i))

            statData(IDX_FILE_COUNT) = _
                CLng(statData(IDX_FILE_COUNT)) + 1

            statData(IDX_TOTAL_BYTES) = _
                CDbl(statData(IDX_TOTAL_BYTES)) + fileSize

            '長期未更新：最長の設定年数以上
            If yearsOld >= gOldYear3 Then

                statData(IDX_OLD_COUNT) = _
                    CLng(statData(IDX_OLD_COUNT)) + 1

                statData(IDX_OLD_BYTES) = _
                    CDbl(statData(IDX_OLD_BYTES)) + fileSize

            End If

            If CStr( _
                GetFileValue(FC_CLEANUP, i)) = "○" Then

                statData(IDX_CLEANUP_COUNT) = _
                    CLng(statData(IDX_CLEANUP_COUNT)) + 1

                statData(IDX_CLEANUP_BYTES) = _
                    CDbl(statData(IDX_CLEANUP_BYTES)) + fileSize

            End If

            If CStr( _
                GetFileValue(FC_DUPLICATE, i)) = "○" Then

                statData(IDX_DUP_COUNT) = _
                    CLng(statData(IDX_DUP_COUNT)) + 1

                statData(IDX_DUP_BYTES) = _
                    CDbl(statData(IDX_DUP_BYTES)) + fileSize

            End If

            '大容量×長期未更新は、整理候補ロジックと合わせて
            '「判定2年以上」かつ「大容量」とする。
            If yearsOld >= gOldYear2 And _
               CStr(GetFileValue(FC_LARGE, i)) = "○" Then

                statData(IDX_LARGE_OLD_COUNT) = _
                    CLng(statData(IDX_LARGE_OLD_COUNT)) + 1

                statData(IDX_LARGE_OLD_BYTES) = _
                    CDbl(statData(IDX_LARGE_OLD_BYTES)) + fileSize

            End If

            If CStr(GetFileValue(FC_NAMEWARN, i)) = "○" Or _
               CStr(GetFileValue(FC_PATHWARN, i)) = "○" Then

                statData(IDX_WARN_COUNT) = _
                    CLng(statData(IDX_WARN_COUNT)) + 1

            End If

            stats(parentPath) = statData

        End If

    Next i

    '------------------------------------------------------
    ' 子フォルダの値を親へ伝播
    '------------------------------------------------------
    For level = maxLevel To 1 Step -1

        For Each key In keys

            folderData = folders(key)

            If CLng(folderData(4)) = level Then

                parentPath = CStr(folderData(12))

                If Len(parentPath) > 0 Then

                    If stats.Exists(parentPath) Then

                        statData = stats(CStr(key))
                        parentData = stats(parentPath)

                        For i = 1 To STAT_COUNT

                            If i = IDX_TOTAL_BYTES Or _
                               i = IDX_OLD_BYTES Or _
                               i = IDX_CLEANUP_BYTES Or _
                               i = IDX_DUP_BYTES Or _
                               i = IDX_LARGE_OLD_BYTES Then

                                parentData(i) = _
                                    CDbl(parentData(i)) + _
                                    CDbl(statData(i))

                            Else

                                parentData(i) = _
                                    CLng(parentData(i)) + _
                                    CLng(statData(i))

                            End If

                        Next i

                        stats(parentPath) = parentData

                    End If

                End If

            End If

        Next key

    Next level

    '------------------------------------------------------
    ' スコア算出
    '------------------------------------------------------
    For Each key In keys

        statData = stats(CStr(key))

        countRatio = SafeRatio( _
            CDbl(statData(IDX_OLD_COUNT)), _
            CDbl(statData(IDX_FILE_COUNT)))

        byteRatio = SafeRatio( _
            CDbl(statData(IDX_OLD_BYTES)), _
            CDbl(statData(IDX_TOTAL_BYTES)))

        severityOld = _
            NormalizeHealthRatio( _
                countRatio * 0.6 + _
                byteRatio * 0.4, _
                LIMIT_OLD)

        countRatio = SafeRatio( _
            CDbl(statData(IDX_CLEANUP_COUNT)), _
            CDbl(statData(IDX_FILE_COUNT)))

        byteRatio = SafeRatio( _
            CDbl(statData(IDX_CLEANUP_BYTES)), _
            CDbl(statData(IDX_TOTAL_BYTES)))

        severityCleanup = _
            NormalizeHealthRatio( _
                countRatio * 0.6 + _
                byteRatio * 0.4, _
                LIMIT_CLEANUP)

        countRatio = SafeRatio( _
            CDbl(statData(IDX_DUP_COUNT)), _
            CDbl(statData(IDX_FILE_COUNT)))

        byteRatio = SafeRatio( _
            CDbl(statData(IDX_DUP_BYTES)), _
            CDbl(statData(IDX_TOTAL_BYTES)))

        If gHashEnabled Then

            severityDup = _
                NormalizeHealthRatio( _
                    countRatio * 0.4 + _
                    byteRatio * 0.6, _
                    LIMIT_DUP)

        Else

            'ハッシュ重複検出OFF時は重複を減点しない。
            severityDup = 0#

        End If

        countRatio = SafeRatio( _
            CDbl(statData(IDX_LARGE_OLD_COUNT)), _
            CDbl(statData(IDX_FILE_COUNT)))

        byteRatio = SafeRatio( _
            CDbl(statData(IDX_LARGE_OLD_BYTES)), _
            CDbl(statData(IDX_TOTAL_BYTES)))

        severityLargeOld = _
            NormalizeHealthRatio( _
                countRatio * 0.4 + _
                byteRatio * 0.6, _
                LIMIT_LARGE_OLD)

        countRatio = SafeRatio( _
            CDbl(statData(IDX_WARN_COUNT)), _
            CDbl(statData(IDX_FILE_COUNT)))

        severityWarn = _
            NormalizeHealthRatio( _
                countRatio, _
                LIMIT_WARN)

        countRatio = SafeRatio( _
            CDbl(statData(IDX_EMPTY_FOLDER_COUNT)), _
            CDbl(statData(IDX_FOLDER_COUNT)))

        severityEmpty = _
            NormalizeHealthRatio( _
                countRatio, _
                LIMIT_EMPTY)

        penaltyOld = severityOld * gHealthWeightOld
        penaltyCleanup = severityCleanup * gHealthWeightCleanup
        penaltyDup = severityDup * gHealthWeightDuplicate
        penaltyLargeOld = severityLargeOld * gHealthWeightLargeOld
        penaltyWarn = severityWarn * gHealthWeightWarning
        penaltyEmpty = severityEmpty * gHealthWeightEmpty

        score = _
            100# - _
            penaltyOld - _
            penaltyCleanup - _
            penaltyDup - _
            penaltyLargeOld - _
            penaltyWarn - _
            penaltyEmpty

        If score < 0# Then score = 0#
        If score > 100# Then score = 100#

        score = Round(score, 1)
        rank = GetHealthRank(score)

        reason = BuildHealthReason( _
            penaltyOld, _
            penaltyCleanup, _
            penaltyDup, _
            penaltyLargeOld, _
            penaltyWarn, _
            penaltyEmpty)

        If Not gHashEnabled Then

            If Len(reason) > 0 Then
                reason = reason & " / "
            End If

            reason = reason & "※重複評価OFF"

        End If

        healthData(1) = score
        healthData(2) = rank
        healthData(3) = reason

        result.Add CStr(key), healthData

    Next key

    Set BuildFolderHealthData = result

End Function


Private Function SafeRatio( _
    ByVal numerator As Double, _
    ByVal denominator As Double) As Double

    If denominator <= 0# Then
        SafeRatio = 0#
    Else
        SafeRatio = numerator / denominator
    End If

End Function


Private Function NormalizeHealthRatio( _
    ByVal actualRatio As Double, _
    ByVal badLimit As Double) As Double

    If actualRatio <= 0# Then
        NormalizeHealthRatio = 0#
        Exit Function
    End If

    If badLimit <= 0# Then
        NormalizeHealthRatio = 1#
        Exit Function
    End If

    NormalizeHealthRatio = _
        actualRatio / badLimit

    If NormalizeHealthRatio > 1# Then
        NormalizeHealthRatio = 1#
    End If

End Function


Private Function GetHealthRank( _
    ByVal score As Double) As String

    Select Case score

        Case Is >= 90#
            GetHealthRank = "A"

        Case Is >= 75#
            GetHealthRank = "B"

        Case Is >= 60#
            GetHealthRank = "C"

        Case Is >= 40#
            GetHealthRank = "D"

        Case Else
            GetHealthRank = "E"

    End Select

End Function


Private Function BuildHealthReason( _
    ByVal penaltyOld As Double, _
    ByVal penaltyCleanup As Double, _
    ByVal penaltyDup As Double, _
    ByVal penaltyLargeOld As Double, _
    ByVal penaltyWarn As Double, _
    ByVal penaltyEmpty As Double) As String

    Dim names(1 To 6) As String
    Dim values(1 To 6) As Double

    Dim i As Long
    Dim j As Long
    Dim maxIndex As Long

    Dim tempName As String
    Dim tempValue As Double
    Dim result As String

    names(1) = "長期未更新"
    values(1) = penaltyOld

    names(2) = "整理候補"
    values(2) = penaltyCleanup

    names(3) = "重複"
    values(3) = penaltyDup

    names(4) = "大容量×長期未更新"
    values(4) = penaltyLargeOld

    names(5) = "パス／ファイル名警告"
    values(5) = penaltyWarn

    names(6) = "空フォルダ"
    values(6) = penaltyEmpty

    '減点の大きい順に並べる。
    For i = 1 To 5

        maxIndex = i

        For j = i + 1 To 6

            If values(j) > values(maxIndex) Then
                maxIndex = j
            End If

        Next j

        If maxIndex <> i Then

            tempValue = values(i)
            values(i) = values(maxIndex)
            values(maxIndex) = tempValue

            tempName = names(i)
            names(i) = names(maxIndex)
            names(maxIndex) = tempName

        End If

    Next i

    '上位3要因を表示。減点0は表示しない。
    For i = 1 To 3

        If values(i) > 0# Then

            If Len(result) > 0 Then
                result = result & " / "
            End If

            result = _
                result & _
                names(i) & _
                " -" & _
                Format$(values(i), "0.0")

        End If

    Next i

    If Len(result) = 0 Then
        result = "減点要因なし"
    End If

    BuildHealthReason = result

End Function


'==========================================================
' ErrorLog出力
'==========================================================
Private Sub OutputErrorLog()

    Dim ws As Worksheet
    Dim data() As Variant

    Dim errorCount As Long
    Dim r As Long
    Dim c As Long

    Set ws = ThisWorkbook.Worksheets(SHEET_ERROR)

    DeleteExistingTable ws, TABLE_ERROR

    ws.Range( _
        "A2:E" & _
        ws.Rows.Count).ClearContents

    errorCount = GetErrorCount()

    If errorCount = 0 Then Exit Sub

    ReDim data( _
        1 To errorCount, _
        1 To 5)

    For r = 1 To errorCount

        For c = 1 To 5
            data(r, c) = _
                GetErrorValue(c, r)
        Next c

    Next r

    ws.Range("A2").Resize( _
        errorCount, _
        5).value = data

    ws.Columns("A").NumberFormat = _
        "yyyy/mm/dd hh:mm:ss"

    CreateTable _
        ws, _
        ws.Range("A1").Resize( _
            errorCount + 1, _
            5), _
        TABLE_ERROR

End Sub


'==========================================================
' テーブル
'==========================================================
Private Sub DeleteExistingTable( _
    ByVal ws As Worksheet, _
    ByVal tableName As String)

    Dim lo As ListObject
    Dim oldDataRange As Range

    If ws Is Nothing Then Exit Sub
    If Len(tableName) = 0 Then Exit Sub

    Set lo = Nothing
    Set oldDataRange = Nothing

    On Error Resume Next
    Set lo = ws.ListObjects(tableName)
    On Error GoTo 0

    If Not lo Is Nothing Then

        '--------------------------------------------------
        ' Unlistするとテーブル自体は解除されるが、
        ' 旧テーブルの書式（縞模様・罫線など）は
        ' 通常セル側に残る。
        '
        ' 大きいスキャンの後に小さいスキャンを行うと、
        ' この残存書式のせいで旧テーブル範囲まで
        ' テーブルが続いているように見えるため、
        ' 旧データ部分の書式を明示的に消す。
        '--------------------------------------------------
        If Not lo.DataBodyRange Is Nothing Then
            Set oldDataRange = lo.DataBodyRange
        End If

        lo.Unlist

        If Not oldDataRange Is Nothing Then
            oldDataRange.ClearFormats
        End If

    End If

End Sub


Private Sub CreateTable( _
    ByVal ws As Worksheet, _
    ByVal sourceRange As Range, _

    Dim lo As ListObject

    Set lo = _
        ws.ListObjects.Add( _
            xlSrcRange, _
            sourceRange, _
            , _
            xlYes)

    lo.Name = tableName
    lo.TableStyle = "TableStyleMedium2"

End Sub


'==========================================================
' Dashboard更新
'==========================================================
Private Sub UpdateDashboard()

    Dim ws As Worksheet
    Dim i As Long

    Dim totalSize As Double
    Dim cleanupSize As Double
    Dim yearsOld As Double

    Dim old1 As Long
    Dim old2 As Long
    Dim old3 As Long

    Dim largeCount As Long
    Dim duplicateCount As Long
    Dim cleanupCount As Long
    Dim emptyFolderCount As Long

    Dim folders As Object
    Dim key As Variant
    Dim folderData As Variant

    Set ws = _
        ThisWorkbook.Worksheets(SHEET_DASHBOARD)

    For i = 1 To GetFileCount()

        totalSize = _
            totalSize + _
            CDbl(GetFileValue(FC_SIZE, i))

        yearsOld = _
            CDbl(GetFileValue(FC_YEARS, i))

        If yearsOld >= gOldYear1 Then
            old1 = old1 + 1
        End If

        If yearsOld >= gOldYear2 Then
            old2 = old2 + 1
        End If

        If yearsOld >= gOldYear3 Then
            old3 = old3 + 1
        End If

        If CStr( _
            GetFileValue(FC_LARGE, i)) = "○" Then

            largeCount = largeCount + 1

        End If

        If CStr( _
            GetFileValue(FC_DUPLICATE, i)) = "○" Then

            duplicateCount = duplicateCount + 1

        End If

        If CStr( _
            GetFileValue(FC_CLEANUP, i)) = "○" Then

            cleanupCount = cleanupCount + 1

            cleanupSize = _
                cleanupSize + _
                CDbl(GetFileValue(FC_SIZE, i))

        End If

    Next i

    Set folders = GetFolderDictionary()

    If Not folders Is Nothing Then

        For Each key In folders.keys

            folderData = folders(key)

            If CBool(folderData(10)) Then
                emptyFolderCount = _
                    emptyFolderCount + 1
            End If

        Next key

    End If

    ws.Range("C10").value = gScanRoot
    ws.Range("C11").value = gScanEnd

    ws.Range("C11").NumberFormat = _
        "yyyy/mm/dd hh:mm:ss"

    ws.Range("C12").value = _
        FormatElapsedSeconds( _
            DateDiff( _
                "s", _
                gScanStart, _
                gScanEnd))

    SetCardValue _
        ws, "B14", _
        Format$(GetFileCount(), "#,##0")

    SetCardValue _
        ws, "E14", _
        Format$(GetFolderCount(), "#,##0")

    SetCardValue _
        ws, "H14", _
        FormatFileSize(totalSize)

    SetCardValue _
        ws, "B18", _
        Format$(old1, "#,##0")

    SetCardValue _
        ws, "E18", _
        Format$(old2, "#,##0")

    SetCardValue _
        ws, "H18", _
        Format$(old3, "#,##0")

    SetCardValue _
        ws, "B22", _
        Format$(largeCount, "#,##0")

    SetCardValue _
        ws, "E22", _
        Format$(duplicateCount, "#,##0")

    SetCardValue _
        ws, "H22", _
        Format$(emptyFolderCount, "#,##0")

    SetCardValue _
        ws, "B26", _
        Format$(cleanupCount, "#,##0")

    SetCardValue _
        ws, "E26", _
        FormatFileSize(cleanupSize)

    SetCardValue _
        ws, "H26", _
        Format$(GetErrorCount(), "#,##0")

End Sub


Private Sub SetCardValue( _
    ByVal ws As Worksheet, _
    ByVal cardTopLeft As String, _
    ByVal valueText As String)

    Dim currentValue As String
    Dim p As Long
    Dim titleText As String

    currentValue = _
        CStr(ws.Range(cardTopLeft).value)

    p = InStr(currentValue, vbLf)

    If p > 0 Then
        titleText = _
            Left$(currentValue, p - 1)
    Else
        titleText = currentValue
    End If

    ws.Range(cardTopLeft).value = _
        titleText & vbLf & valueText

End Sub


Private Function FormatElapsedSeconds( _
    ByVal totalSeconds As Long) As String

    Dim hours As Long
    Dim minutes As Long
    Dim seconds As Long

    hours = totalSeconds \ 3600

    minutes = _
        (totalSeconds Mod 3600) \ 60

    seconds = _
        totalSeconds Mod 60

    FormatElapsedSeconds = _
        Format$(hours, "00") & ":" & _
        Format$(minutes, "00") & ":" & _
        Format$(seconds, "00")

End Function

'==========================================================
' 共通フィルター
'==========================================================
Private Sub ClearTableFilter( _
    ByVal lo As ListObject)

    On Error Resume Next

    If lo.AutoFilter.FilterMode Then
        lo.AutoFilter.ShowAllData
    End If

    On Error GoTo 0

End Sub


Private Function GetVisibleTableRowCount( _
    ByVal lo As ListObject) As Long

    Dim area As Range
    Dim visibleRange As Range

    If lo.DataBodyRange Is Nothing Then
        Exit Function
    End If

    On Error Resume Next

    Set visibleRange = _
        lo.DataBodyRange.Columns(1). _
        SpecialCells(xlCellTypeVisible)

    On Error GoTo 0

    If visibleRange Is Nothing Then
        Exit Function
    End If

    For Each area In visibleRange.Areas

        GetVisibleTableRowCount = _
            GetVisibleTableRowCount + _
            area.Rows.Count

    Next area

End Function

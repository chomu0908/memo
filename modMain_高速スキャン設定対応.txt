'==========================================================

Public Sub SetupTool()

    On Error GoTo EH

    Application.ScreenUpdating = False
    Application.EnableEvents = False

    EnsureSheet SHEET_DASHBOARD
    EnsureSheet SHEET_FILELIST
    EnsureSheet SHEET_FOLDER
    EnsureSheet SHEET_ERROR
    EnsureSheet SHEET_SETTINGS

    SetupSettingsSheet

    LoadSettings

    SetupFileListSheet
    SetupFolderSummarySheet
    SetupErrorLogSheet
    SetupDashboardSheet

    ThisWorkbook.Worksheets( _
        SHEET_DASHBOARD).Activate

    MsgBox _
        "ファイル棚卸しツールのセットアップが完了しました。", _
        vbInformation, _
        "SetupTool"

ExitPoint:

    Application.ScreenUpdating = True
    Application.EnableEvents = True

    Exit Sub

EH:

    MsgBox _
        "セットアップ中にエラーが発生しました。" & _
        vbCrLf & _
        Err.Number & _
        " : " & _
        Err.Description, _
        vbExclamation, _
        "SetupTool"

    Resume ExitPoint

End Sub


Private Sub EnsureSheet( _
    ByVal sheetName As String)

    Dim ws As Worksheet

    On Error Resume Next

    Set ws = _
        ThisWorkbook.Worksheets(sheetName)

    On Error GoTo 0

    If ws Is Nothing Then

        Set ws = _
            ThisWorkbook.Worksheets.Add( _
                After:=ThisWorkbook.Worksheets( _
                    ThisWorkbook.Worksheets.Count))

        ws.Name = sheetName

    End If

End Sub

Public Sub LoadSettings()

    Dim ws As Worksheet
    Dim totalHealthWeight As Double

    Set ws = _
        ThisWorkbook.Worksheets( _
            SHEET_SETTINGS)

    '古いブックでもStartScan時に設定を自動補完する。
    EnsureFastScanSettingDefault ws
    EnsureHealthSettingsDefaults ws

    gLargeFileMB = CDbl(ws.Range("B2").Value)
    gOldYear1 = CLng(ws.Range("B3").Value)
    gOldYear2 = CLng(ws.Range("B4").Value)
    gOldYear3 = CLng(ws.Range("B5").Value)
    gFileNameWarnLength = CLng(ws.Range("B6").Value)
    gPathWarnLength = CLng(ws.Range("B7").Value)

    gHashEnabled = _
        (UCase$(Trim$(CStr(ws.Range("B8").Value))) = "ON")

    gFastScanEnabled = _
        (UCase$(Trim$(CStr(ws.Range("B9").Value))) = "ON")

    gHealthWeightOld = CDbl(ws.Range("B11").Value)
    gHealthWeightCleanup = CDbl(ws.Range("B12").Value)
    gHealthWeightDuplicate = CDbl(ws.Range("B13").Value)
    gHealthWeightLargeOld = CDbl(ws.Range("B14").Value)
    gHealthWeightWarning = CDbl(ws.Range("B15").Value)
    gHealthWeightEmpty = CDbl(ws.Range("B16").Value)

    If gLargeFileMB < 0 Then
        Err.Raise _
            vbObjectError + 100, _
            , _
            "大容量判定サイズが不正です。"
    End If

    If gOldYear1 < 0 Or _
       gOldYear2 < 0 Or _
       gOldYear3 < 0 Then

        Err.Raise _
            vbObjectError + 101, _
            , _
            "長期未更新年数が不正です。"
    End If

    If gOldYear1 > gOldYear2 Or _
       gOldYear2 > gOldYear3 Then

        Err.Raise _
            vbObjectError + 102, _
            , _
            "長期未更新年数は、判定1 <= 判定2 <= 判定3 の順に設定してください。"
    End If

    If gHealthWeightOld < 0 Or _
       gHealthWeightCleanup < 0 Or _
       gHealthWeightDuplicate < 0 Or _
       gHealthWeightLargeOld < 0 Or _
       gHealthWeightWarning < 0 Or _
       gHealthWeightEmpty < 0 Then

        Err.Raise _
            vbObjectError + 103, _
            , _
            "健全度の重みは0以上で設定してください。"
    End If

    totalHealthWeight = _
        gHealthWeightOld + _
        gHealthWeightCleanup + _
        gHealthWeightDuplicate + _
        gHealthWeightLargeOld + _
        gHealthWeightWarning + _
        gHealthWeightEmpty

    If Abs(totalHealthWeight - 100#) > 0.0001 Then
        Err.Raise _
            vbObjectError + 104, _
            , _
            "健全度の重み合計は100にしてください。現在の合計: " & _
            Format$(totalHealthWeight, "0.0")
    End If

End Sub

'==========================================================
' スキャン
'==========================================================
Public Sub StartScan()

    Dim selectedFolder As String

    Dim scanSucceeded As Boolean

    Dim errorNumber As Long
    Dim errorDescription As String

    On Error GoTo EH

    selectedFolder = _
        SelectFolder()

    If Len(selectedFolder) = 0 Then
        Exit Sub
    End If

    LoadSettings

    SaveExcelState
    PrepareExcelForScan

    gScanRoot = _
        selectedFolder

    gScanStart = Now

    '------------------------------------------------------
    ' 前回スキャンの終了時刻を確実に消す
    '------------------------------------------------------
    gScanEnd = 0

    ResetScanData

    frmProgress.Show vbModeless
    frmProgress.StartProgress

    DoEvents


    '======================================================
    ' 1. スキャン
    '======================================================
    frmProgress.SetStatus _
        "フォルダを走査しています..."

    Application.StatusBar = _
        "フォルダを走査しています..."

    DoEvents

    ScanFolderTree _
        gScanRoot


    '======================================================
    ' 2. 重複
    '======================================================
    If gHashEnabled Then

        frmProgress.SetStatus _
            "重複候補を確認しています..."

        Application.StatusBar = _
            "重複候補を確認しています..."

    Else

        frmProgress.SetStatus _
            "重複検出はOFFです"

        Application.StatusBar = _
            "重複検出OFF - 次の処理へ進みます"

    End If

    DoEvents

    DetectDuplicateFiles


    '======================================================
    ' 3. フォルダ集計
    '======================================================
    frmProgress.SetStatus _
        "フォルダ情報を集計しています..."

    Application.StatusBar = _
        "フォルダ情報を集計しています..."

    DoEvents

    FinalizeFolderSummary


    '======================================================
    ' 4. 整理候補
    '======================================================
    frmProgress.SetStatus _
        "整理候補を判定しています..."

    Application.StatusBar = _
        "整理候補を判定しています..."

    DoEvents

    FinalizeCleanupCandidates


    '======================================================
    ' ★重要
    '
    ' Dashboard更新ではgScanEndを使用する。
    '
    ' そのためOutputAllResultsより先に
    ' 終了日時を確定させる。
    '
    ' 以前は
    '
    ' OutputAllResults
    ' gScanEnd = Now
    '
    ' だったため、gScanEnd=0の状態で
    ' DateDiff("s", gScanStart, gScanEnd)
    ' が実行され、エラー6（オーバーフロー）が
    ' 発生していた。
    '======================================================
    gScanEnd = Now


    '======================================================
    ' 5. Excel出力
    '======================================================
    frmProgress.SetStatus _
        "Excelへ結果を出力しています..."

    Application.StatusBar = _
        "Excelへ結果を出力しています..."

    DoEvents

    OutputAllResults

    scanSucceeded = True


CleanExit:

    On Error Resume Next

    Unload frmProgress

    RestoreExcelState

    If scanSucceeded Then

        ThisWorkbook.Worksheets( _
            SHEET_DASHBOARD).Activate

        MsgBox _
            "スキャンが完了しました。" & _
            vbCrLf & vbCrLf & _
            "ファイル数 : " & _
            Format$( _
                GetFileCount(), _
                "#,##0") & _
            vbCrLf & _
            "フォルダ数 : " & _
            Format$( _
                GetFolderCount(), _
                "#,##0") & _
            vbCrLf & _
            "エラー件数 : " & _
            Format$( _
                GetErrorCount(), _
                "#,##0"), _
            vbInformation, _
            "ファイル棚卸し"

    End If

    Exit Sub


EH:

    errorNumber = Err.Number
    errorDescription = Err.Description

    On Error Resume Next

    AddScanError _
        "メイン処理", _
        gScanRoot, _
        errorNumber, _
        errorDescription

    '------------------------------------------------------
    ' エラー時もOutputAllResultsより先に終了日時を確定
    '------------------------------------------------------
    gScanEnd = Now

    If GetFileCount() > 0 Then

        frmProgress.SetStatus _
            "取得済みデータを整理しています..."

        DoEvents

        FinalizeFolderSummary
        FinalizeCleanupCandidates

        frmProgress.SetStatus _
            "取得済みデータを出力しています..."

        DoEvents

        OutputAllResults

    End If

    MsgBox _
        "スキャン処理で予期しないエラーが発生しました。" & _
        vbCrLf & _
        errorNumber & _
        " : " & _
        errorDescription & _
        vbCrLf & vbCrLf & _
        "取得済みのデータは可能な範囲で出力しました。", _
        vbExclamation, _
        "ファイル棚卸し"

    Resume CleanExit

End Sub


Private Function SelectFolder() As String

    Dim dlg As FileDialog

    Set dlg = _
        Application.FileDialog( _
            msoFileDialogFolderPicker)

    With dlg

        .Title = _
            "棚卸し対象フォルダを選択してください"

        .AllowMultiSelect = False

        If .Show <> -1 Then

            SelectFolder = _
                vbNullString

            Exit Function

        End If

        SelectFolder = _
            .SelectedItems(1)

    End With

End Function

Private Sub SaveExcelState()

    mOldCalculation = _
        Application.Calculation

    mOldScreenUpdating = _
        Application.ScreenUpdating

    mOldEnableEvents = _
        Application.EnableEvents

    mOldDisplayAlerts = _
        Application.DisplayAlerts

    mOldStatusBar = _
        Application.StatusBar

    mOldCursor = _
        Application.Cursor

End Sub


Private Sub PrepareExcelForScan()

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False

    Application.Calculation = _
        xlCalculationManual

    '待機カーソル固定はしない
    Application.Cursor = xlDefault

    Application.StatusBar = _
        "ファイル棚卸しを開始しています..."

End Sub


Private Sub RestoreExcelState()

    On Error Resume Next

    Application.ScreenUpdating = _
        mOldScreenUpdating

    Application.EnableEvents = _
        mOldEnableEvents

    Application.DisplayAlerts = _
        mOldDisplayAlerts

    Application.Calculation = _
        mOldCalculation

    Application.Cursor = _
        mOldCursor

    Application.StatusBar = _
        mOldStatusBar

End Sub

'==========================================================
Public Sub OpenSelectedInExplorer()

    Select Case ActiveSheet.Name

        Case SHEET_FILELIST

            OpenSelectedFileInExplorer

        Case SHEET_FOLDER

            OpenSelectedFolderInExplorer

        Case Else

            MsgBox _
                "FileListまたはFolderSummaryで対象行を選択してください。", _
                vbInformation, _
                "Explorer"

    End Select

End Sub


Public Sub OpenSelectedFileInExplorer()

    Dim ws As Worksheet
    Dim targetPath As String

    On Error GoTo EH

    Set ws = _
        ThisWorkbook.Worksheets( _
            SHEET_FILELIST)

    If ActiveSheet.Name <> ws.Name Then

        MsgBox _
            "FileListで対象ファイルを選択してください。", _
            vbInformation

        Exit Sub

    End If

    targetPath = _
        GetSelectedTableValue( _
            ws, _
            "フルパス")

    If Len(targetPath) = 0 Then

        MsgBox _
            "ファイル行を選択してください。", _
            vbInformation

        Exit Sub

    End If

    Shell _
        "explorer.exe /select,""" & _
        Replace( _
            targetPath, _
            """", _
            """""") & _
        """", _
        vbNormalFocus

    Exit Sub

EH:

    MsgBox _
        "Explorerを開けませんでした。" & _
        vbCrLf & _
        Err.Description, _
        vbExclamation, _
        "Explorer"

End Sub

Private Function GetSelectedTableValue( _
    ByVal ws As Worksheet, _
    ByVal headerName As String) As String

    Dim lo As ListObject

    Dim tableRow As Long
    Dim columnIndex As Long

    If ws.ListObjects.Count = 0 Then
        Exit Function
    End If

    Set lo = _
        ws.ListObjects(1)

    If lo.DataBodyRange Is Nothing Then
        Exit Function
    End If

    If Intersect( _
        ActiveCell, _
        lo.DataBodyRange) Is Nothing Then

        Exit Function

    End If

    tableRow = _
        ActiveCell.Row - _
        lo.DataBodyRange.Row + 1

    columnIndex = _
        lo.ListColumns( _
            headerName).Index

    GetSelectedTableValue = _
        CStr( _
            lo.DataBodyRange.Cells( _
                tableRow, _
                columnIndex).value)

End Function
